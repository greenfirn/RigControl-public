#=============================================================================================================
# -- rig conf examples...                                                                                   --
# -- algo, pool, wallet, password get added to miner args from conf                                         --
# -- worker name is left to manualy add with wallet or where ever the pool wants it                         --

# -- %WORKER_NAME%           rigs hostname in ARGS, WALLET, PASS, POOL                                      --
# -- %WARTHOG_TARGET%        auto set WARTHOG_TARGET in ARGS (24/32 thread CPU)                             --
# -- %CPU_THREADS%           cpu total threads -1 (and affinity when xmrig) in ARGS see '03-cpu_threads.sh' --
# -- SCREEN_NAME "gpu"         different screen name than miner name                                        --
# custom miner download url                                                                                 --
# -- CUSTOM_MINER_URL ""                                                                                    --
# custom miner binary name                                                                                  --
# -- CUSTOM_MINER ""                                                                                        --
#=============================================================================================================

# -- to update miner version write miner.conf and restart service --

mkdir -p /etc/rigcontrol
tee /etc/rigcontrol/miner.conf > /dev/null <<'EOF'
XMRIG_VERSION "6.25.0"
BZMINER_VERSION "v23.0.2"
WILDRIG_VERSION "0.47.9"
SRBMINER_VERSION "3.0.6"
RIGEL_VERSION "1.23.1"
LOLMINER_VERSION "1.98"
ONEZEROMINER_VERSION "1.7.3"
GMINER_VERSION "3.44"
TEAMREDMINER_VERSION "0.10.21"
TREXMINER_VERSION "0.26.8"
EOF

sudo systemctl restart docker_events_gpu

# -- xmrig write Nosana rig-cpu.conf --

mkdir -p /etc/rigcontrol
tee /etc/rigcontrol/rig-cpu.conf > /dev/null <<'EOF'
TARGET_IMAGE ""
TARGET_NAME "podman"
APPLY_OC "false"
RESET_OC "false"
SCREEN_NAME "cpu"
CUSTOM_MINER_URL ""
CUSTOM_MINER ""
MINER "xmrig"
ALGO "rx/0"
POOL "pool.supportxmr.com:9000"
WALLET "wallet-address"
PASS "%WORKER_NAME%"
ARGS "-t %CPU_THREADS% --tls -k"
EOF

PASS ""

EOF
