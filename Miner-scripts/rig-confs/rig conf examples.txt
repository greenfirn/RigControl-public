##############################################################################################################
# -- rig conf examples...                                                                                   --
# -- algo, pool, wallet, password get added to miner args from the json below                               --
# -- worker name is left to manually add with wallet or wherever the pool wants it                          --
# --                                                                                                        --
# -- Template tokens (substituted when the config is written):                                              --
# -- %WORKER_NAME%       rigs hostname in ARGS, WALLET/template, PASS, POOL/url                             --
# -- %URL%               substitutes the first pool_urls[] entry into miner_config.url, and any             --
# --                     %URL% token elsewhere (e.g. a custom miner's user_config)                          --
# -- %WARTHOG_TARGET%    auto set WARTHOG_TARGET in ARGS (24/32 thread CPU)                                 --
# -- %CPU_THREADS%       cpu total threads -1 (and affinity when xmrig) in ARGS, see '03-cpu_threads.sh'    --
# --                                                                                                        --
# -- slot ("cpu"/"gpu"/"aux") is implied by filename (rig-cpu.json / rig-gpu.json / rig-aux.json)           --
# -- no SCREEN_NAME/SERVICE_TYPE field needed in the json - the file itself is the slot                     --
# -- multiple failover pools: top-level "pool_urls": [...] array, bare host:port, no scheme prefix          --
# --                                                                                                        --
# -- custom miner download url: top-level "miner": "custom" + "miner_alt", miner_config.install_url         --
# -- custom miner binary name:  miner_config.miner (only present when top-level "miner" is "custom")        --
# --                                                                                                        --
# -- json files kept as plain 'tee' (no sudo) so they can be easily overwritten in home folder via ssh      --
# -- if write error, change owner back to user: sudo chown $USER:$USER *.json                               --
##############################################################################################################
# -- to update miner version, write miner.conf and restart service (miner.conf is a separate                --
# -- version-pins file, not a rig-cpu/rig-gpu/rig-aux config, so it stays in the old KEY "value" format)    --
mkdir -p /etc/rigcontrol
tee /etc/rigcontrol/miner.conf > /dev/null <<'EOF'
XMRIG_VERSION "6.25.0"
BZMINER_VERSION "v23.0.2"
WILDRIG_VERSION "0.47.9"
SRBMINER_VERSION "3.0.6"
RIGEL_VERSION "1.23.1"
LOLMINER_VERSION "1.98"
ONEZEROMINER_VERSION "1.7.3"
GMINER_VERSION "3.44"
TEAMREDMINER_VERSION "0.10.21"
TREXMINER_VERSION "0.26.8"
EOF
sudo systemctl restart docker_events_gpu
